################################################################################
# bladeRF 2.0 micro Nios firmware for Nios V
#
# hdl/quartus/build_bladerf.sh configures this project in the Quartus work
# directory after generating the HAL BSP with niosv-bsp:
#
#   cmake -S <this dir> -B <work dir>/bladeRF_nios \
#         -DBSP_DIR=<work dir>/bladeRF_nios_bsp -DCMAKE_BUILD_TYPE=Release
#   cmake --build <work dir>/bladeRF_nios
#
# It produces bladeRF_nios.elf and mem_init/nios_system_ram.hex, the on-chip
# RAM image that the Quartus project picks up through mem_init/meminit.qip.
#
# The Nios II firmware is built by the Makefile in this directory instead.
# Keep the source list and flags below in sync with it and with
# ../../../common/bladerf/software/bladeRF_nios/Makefile.
################################################################################
cmake_minimum_required(VERSION 3.14)

if(NOT BSP_DIR)
    message(FATAL_ERROR "BSP_DIR must point to the niosv-bsp generated BSP")
endif()
get_filename_component(BSP_DIR "${BSP_DIR}" ABSOLUTE)

# C library. newlib-nano is the counterpart of the Nios II "small C library"
# (hal.enable_small_c_library): with the full newlib the firmware is about
# 60 KB too big for the on-chip RAM. newlib-nano's headers differ from the
# full newlib's (struct _reent), so the BSP, libad936x and the firmware are
# all compiled with nano.specs, not only linked with it (at link time it turns
# -lc and -lm into -lc_nano and -lm_nano). picolibc is set up by the BSP
# instead (hal.linker.use_picolibc, see build_bladerf.sh).
set(NIOSV_LIBC "newlib-nano" CACHE STRING "C library: newlib-nano or picolibc")
set_property(CACHE NIOSV_LIBC PROPERTY STRINGS newlib-nano picolibc)
if(NIOSV_LIBC STREQUAL "newlib-nano")
    add_compile_options(--specs=nano.specs)
    add_link_options(--specs=nano.specs)
elseif(NOT NIOSV_LIBC STREQUAL "picolibc")
    message(FATAL_ERROR "NIOSV_LIBC must be newlib-nano or picolibc, not ${NIOSV_LIBC}")
endif()

# Leave out unused functions and data (much of the AD9361 driver's API and
# unused parts of the HAL and C library): about 33 KB of the RAM is then left
# for the stack and heap instead of about 10 KB.
add_compile_options(-ffunction-sections -fdata-sections)
add_link_options(-Wl,--gc-sections)

# Same structure as the CMakeLists.txt that niosv-app generates: the BSP
# provides the hal2_bsp library, toolchain settings and linker script.
add_subdirectory(${BSP_DIR} bladeRF_nios_bsp)
include(${BSP_DIR}/toolchain.cmake)

# Treat the BSP headers as system headers, like any vendor SDK's: the firmware
# is built with -Wextra -Werror, which the HAL headers of Quartus 25.1 do not
# pass (intel_niosv.h: "type qualifiers ignored on function return type").
get_target_property(_bsp_includes ${BspLibraryName} INTERFACE_INCLUDE_DIRECTORIES)
set_property(TARGET ${BspLibraryName} APPEND PROPERTY
             INTERFACE_SYSTEM_INCLUDE_DIRECTORIES ${_bsp_includes})

project(bladeRF_nios C ASM)

get_filename_component(BLADERF_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.."
                       ABSOLUTE)
set(COMMON_DIR "${BLADERF_ROOT}/hdl/fpga/platforms/common/bladerf/software/bladeRF_nios")

# On-chip RAM the firmware runs from, as generated by Qsys into system.h
set(NIOS_RAM_NAME "nios_system_ram")
file(STRINGS "${BSP_DIR}/system.h" _ram_base REGEX "^#define RAM_BASE ")
file(STRINGS "${BSP_DIR}/system.h" _ram_span REGEX "^#define RAM_SPAN ")
string(REGEX REPLACE "^#define RAM_BASE +" "" RAM_BASE "${_ram_base}")
string(REGEX REPLACE "^#define RAM_SPAN +" "" RAM_SPAN "${_ram_span}")
if(NOT RAM_BASE OR NOT RAM_SPAN)
    message(FATAL_ERROR "RAM_BASE/RAM_SPAN not found in ${BSP_DIR}/system.h")
endif()
math(EXPR RAM_END "${RAM_BASE} + ${RAM_SPAN} - 1" OUTPUT_FORMAT HEXADECIMAL)
message(STATUS "Nios V RAM: base ${RAM_BASE}, ${RAM_SPAN} bytes")

# AD936x driver (ADI no-OS, patched), built with the same toolchain and flags
add_subdirectory(${BLADERF_ROOT}/hdl/fpga/ip/analogdevicesinc/no_OS libad936x)

add_executable(bladeRF_nios.elf
    # bladeRF 2.0 micro sources (Makefile in this directory)
    src/bladeRF_nios.c
    ${BLADERF_ROOT}/fpga_common/src/ad936x_helpers.c
    ${BLADERF_ROOT}/fpga_common/src/bladerf2_common.c
    ${COMMON_DIR}/src/devices_rfic.c
    ${COMMON_DIR}/src/devices_rfic_cmds.c
    ${COMMON_DIR}/src/devices_rfic_queue.c
    ${COMMON_DIR}/src/pkt_retune2.c
    ${BLADERF_ROOT}/host/common/src/range.c

    # Common sources (common bladeRF_nios Makefile)
    ${COMMON_DIR}/src/devices.c
    ${COMMON_DIR}/src/pkt_8x8.c
    ${COMMON_DIR}/src/pkt_8x16.c
    ${COMMON_DIR}/src/pkt_8x32.c
    ${COMMON_DIR}/src/pkt_8x64.c
    ${COMMON_DIR}/src/pkt_16x64.c
    ${COMMON_DIR}/src/pkt_32x32.c
    ${COMMON_DIR}/src/pkt_legacy.c
    ${COMMON_DIR}/src/devices_sim.c
)

target_include_directories(bladeRF_nios.elf PRIVATE
    src
    ${COMMON_DIR}/src
    ${BLADERF_ROOT}/firmware_common
    ${BLADERF_ROOT}/fpga_common/include
    ${BLADERF_ROOT}/host/libraries/libbladeRF/include
    ${BLADERF_ROOT}/host/common/include
    ${BLADERF_ROOT}/thirdparty/analogdevicesinc/no-OS_local/platform_bladerf2
)

target_compile_definitions(bladeRF_nios.elf PRIVATE
    BLADERF_NIOS_BUILD
    NUAND_MODIFICATIONS
    BOARD_BLADERF_MICRO
)

target_compile_options(bladeRF_nios.elf PRIVATE
    -Os -fshort-enums -g
    -Wall -Wextra -Wno-unused-parameter -Werror
)

target_link_options(bladeRF_nios.elf PRIVATE
    "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/bladeRF_nios.map"
)

target_link_libraries(bladeRF_nios.elf PRIVATE
    -T "${BspLinkerScript}" -nostdlib
    ad936x
    "${ExtraArchiveLibraries}"
    -Wl,--start-group "${BspLibraryName}" -lc -lgcc -lm -Wl,--end-group
)

# Disassembly, for debugging
add_custom_command(
    OUTPUT bladeRF_nios.objdump
    DEPENDS bladeRF_nios.elf
    COMMAND "${ToolchainObjdump}" ${ToolchainObjdumpFlags} bladeRF_nios.elf
            > bladeRF_nios.objdump
    COMMENT "Creating bladeRF_nios.objdump"
    VERBATIM
)
add_custom_target(objdump ALL DEPENDS bladeRF_nios.objdump)

# Report how much RAM is left for the stack and heap
add_custom_target(stack-report ALL
    COMMAND niosv-stack-report -p "${ToolchainPrefix}" bladeRF_nios.elf
    DEPENDS bladeRF_nios.elf
    VERBATIM
)

# On-chip RAM initialization file for Quartus, in the same place and format
# (32-bit words, 4-byte records) as the Nios II flow's mem_init_generate
set(MEM_INIT_DIR "${CMAKE_CURRENT_BINARY_DIR}/mem_init")
add_custom_command(
    OUTPUT ${MEM_INIT_DIR}/${NIOS_RAM_NAME}.hex
    DEPENDS bladeRF_nios.elf
    COMMAND ${CMAKE_COMMAND} -E make_directory ${MEM_INIT_DIR}
    COMMAND elf2hex --input=bladeRF_nios.elf
                    --output=${MEM_INIT_DIR}/${NIOS_RAM_NAME}.hex
                    --base=${RAM_BASE} --end=${RAM_END}
                    --width=32 --record=4
    COMMENT "Creating ${MEM_INIT_DIR}/${NIOS_RAM_NAME}.hex"
    VERBATIM
)
add_custom_target(mem_init ALL DEPENDS ${MEM_INIT_DIR}/${NIOS_RAM_NAME}.hex)

# Lets Quartus find the .hex file next to it (same as the Nios II flow)
file(WRITE ${MEM_INIT_DIR}/meminit.qip
     "set_global_assignment -name SEARCH_PATH $::quartus(qip_path)\n")
